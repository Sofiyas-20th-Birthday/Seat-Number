    <script>
      // Edit this list to use your real guest names and seat numbers.
      const guests = [
        { name: "Alina Suleymenova", seat: "14", table: "Table 2" },
        { name: "Sofia Suleymenova", seat: "15", table: "Table 2" },
        { name: "Timur Selyemov", seat: "16", table: "Table 2" },

        { name: "Freja Nielsen", seat: "30", table: "Table 6" },
        { name: "André Silva", seat: "31", table: "Table 6" },
        { name: "Mila Jovanović", seat: "32", table: "Table 6" },
        { name: "Noémi Nagy", seat: "33", table: "Table 7" },
        { name: "Daan de Vries", seat: "34", table: "Table 7" },
        { name: "Elin Andersson", seat: "35", table: "Table 7" },
        { name: "Theo Papadopoulos", seat: "36", table: "Table 7" },
        { name: "Aoife O'Connor", seat: "37", table: "Table 8" }
      ];

      const input = document.querySelector("#guest-search");
      const results = document.querySelector("#results");
      const clearButton = document.querySelector(".clear-button");

      const normalize = (value) =>
        value
          .toLocaleLowerCase()
          .normalize("NFD")
          .replace(/[\u0300-\u036f]/g, "")
          .trim();

      function scoreName(name, query) {
        const target = normalize(name);
        const words = target.split(/\s+/);

        if (target === query) return 1000;
        if (target.startsWith(query)) return 800 - target.length;
        if (words.some((word) => word.startsWith(query))) return 650 - target.length;

        const index = target.indexOf(query);
        if (index !== -1) return 500 - index * 10 - target.length;

        let cursor = 0;
        for (const character of target) {
          if (character === query[cursor]) cursor += 1;
          if (cursor === query.length) return 250 - target.length;
        }

        return 0;
      }

      function safeHighlightedName(name, query) {
        const container = document.createElement("span");
        const normalizedName = normalize(name);
        const index = normalizedName.indexOf(query);

        if (index === -1) {
          container.textContent = name;
          return container.innerHTML;
        }

        container.append(document.createTextNode(name.slice(0, index)));
        const mark = document.createElement("mark");
        mark.textContent = name.slice(index, index + query.length);
        container.append(mark, document.createTextNode(name.slice(index + query.length)));
        return container.innerHTML;
      }

      function showPrompt() {
        results.innerHTML = `
          <div class="prompt-state">
            <div class="prompt-mark" aria-hidden="true">A</div>
            <h2 class="state-title">Your place is waiting</h2>
            <p class="state-copy">Try a first name, surname, or just the first few letters.</p>
          </div>
        `;
      }

      function showNoResults() {
        results.innerHTML = `
          <div class="empty-state">
            <div class="prompt-mark" aria-hidden="true">?</div>
            <h2 class="state-title">We couldn’t find that name</h2>
            <p class="state-copy">Check the spelling or ask the welcome team for a little help.</p>
          </div>
        `;
      }

      function selectGuest(guest) {
        results.innerHTML = `
          <div class="confirmation">
            <div>
              <p class="confirmed-label">We found you</p>
              <h2 class="confirmed-name">${guest.name}</h2>
              <p class="seat-number">${guest.seat}</p>
              <p class="seat-caption">Seat · ${guest.table}</p>
              <button class="back-button" type="button">Search another name</button>
            </div>
          </div>
        `;

        results.querySelector(".back-button").addEventListener("click", () => {
          input.value = "";
          clearButton.classList.remove("visible");
          showPrompt();
          input.focus();
        });
      }

      function updateResults() {
        const query = normalize(input.value);
        clearButton.classList.toggle("visible", Boolean(input.value));

        if (!query) {
          showPrompt();
          return;
        }

        const matches = guests
          .map((guest) => ({ guest, score: scoreName(guest.name, query) }))
          .filter((item) => item.score > 0)
          .sort((a, b) => b.score - a.score || a.guest.name.localeCompare(b.guest.name))
          .slice(0, 3);

        if (!matches.length) {
          showNoResults();
          return;
        }

        results.innerHTML = `
          <ul class="match-list" aria-label="Matching guests">
            ${matches
              .map(
                ({ guest }, index) => `
                  <li>
                    <button class="match ${index === 0 ? "best" : ""}" type="button" data-seat="${guest.seat}">
                      <span>
                        <span class="match-name">${safeHighlightedName(guest.name, query)}</span>
                        <span class="match-detail">${index === 0 ? "Best match" : "Possible match"} · ${guest.table}</span>
                      </span>
                      <span class="seat">
                        <small>Seat</small>
                        <strong>${guest.seat}</strong>
                      </span>
                    </button>
                  </li>
                `
              )
              .join("")}
          </ul>
        `;

        results.querySelectorAll(".match").forEach((button) => {
          button.addEventListener("click", () => {
            const guest = guests.find((item) => item.seat === button.dataset.seat);
            if (guest) selectGuest(guest);
          });
        });
      }

      input.addEventListener("input", updateResults);
      input.addEventListener("keydown", (event) => {
        if (event.key === "Enter") {
          const bestMatch = results.querySelector(".match");
          if (bestMatch) bestMatch.click();
        }
      });

      clearButton.addEventListener("click", () => {
        input.value = "";
        updateResults();
        input.focus();
      });

      showPrompt();
    </script>
